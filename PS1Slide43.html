Direction Arrows on Streamlines
# Add one arrow along each streamline
for level_segs in contours.allsegs:
    for seg in level_segs:
        verts = seg
        if len(verts) < 2: continue
        i = len(verts) // 2
        x0, y0 = verts[i]
        x1, y1 = verts[i+1]
        ax.annotate(
            '', xy=(x1, y1), xytext=(x0, y0),
            arrowprops=dict(
                arrowstyle='->',
                color='black',
                lw=1.8
            )
        )
