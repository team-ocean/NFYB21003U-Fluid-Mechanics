Initial conditions

def fill_reflective_ghosts(hg, hug): # Reflective BC
    hg[0]  = hg[1]
    hg[-1] = hg[-2]
    hug[0]  = -hug[1]
    hug[-1] = -hug[-2]
    return hg, hug
